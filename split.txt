import os

import cv2
from insightface.app import FaceAnalysis
from tqdm import tqdm

OUTPUT_DIR = 'D:/Projects/Study/School/Nam3_Ky2/KHDL/valid'
INPUT_DIR = 'D:/Projects/Study/School/Nam3_Ky2/KHDL/val'

FACE_SIZE = 224

app = FaceAnalysis(
    name='buffalo_l',
    providers=['CUDAExecutionProvider', 'CPUExecutionProvider']
)

app.prepare(ctx_id=0)

os.makedirs(OUTPUT_DIR, exist_ok=True)

person_folders = sorted(os.listdir(INPUT_DIR))

total_faces = 0
total_classes = 0

for person_name in tqdm(person_folders):

    person_path = os.path.join(
        INPUT_DIR,
        person_name
    )

    if not os.path.isdir(person_path):
        continue

    output_person_path = os.path.join(
        OUTPUT_DIR,
        person_name
    )

    os.makedirs(output_person_path, exist_ok=True)

    images = [
        f for f in os.listdir(person_path)
        if f.lower().endswith(('.jpg', '.jpeg', '.png'))
    ]

    saved_count = 0

    for image_name in images:

        image_path = os.path.join(
            person_path,
            image_name
        )

        image = cv2.imread(image_path)

        if image is None:
            continue

        faces = app.get(image)

        if len(faces) == 0:
            continue

        largest_face = max(
            faces,
            key=lambda x: (
                (x.bbox[2] - x.bbox[0]) *
                (x.bbox[3] - x.bbox[1])
            )
        )

        x1, y1, x2, y2 = largest_face.bbox.astype(int)

        # clamp
        h, w = image.shape[:2]

        x1 = max(0, x1)
        y1 = max(0, y1)
        x2 = min(w, x2)
        y2 = min(h, y2)

        face_crop = image[y1:y2, x1:x2]

        if face_crop.size == 0:
            continue

        face_crop = cv2.resize(
            face_crop,
            (FACE_SIZE, FACE_SIZE)
        )

        save_path = os.path.join(
            output_person_path,
            image_name
        )

        cv2.imwrite(save_path, face_crop)

        saved_count += 1
        total_faces += 1

    if saved_count > 0:
        total_classes += 1

print(f'Total classes: {total_classes}')
print(f'Total faces  : {total_faces}')
